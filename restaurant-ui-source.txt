import { Suspense } from "react";
import { connection } from "next/server";
import Link from "next/link";
import { requireAnyRole } from "@/lib/auth/authorization";
import { createClient } from "@/lib/supabase/server";
import RestaurantManagement from "./restaurant-management";
import HiringManagement from "./hiring-management";

type RestaurantStatus = "pending" | "active" | "suspended" | "rejected";
type VerificationStatus = "unverified" | "pending" | "verified" | "rejected";
type MembershipStatus = "invited" | "active" | "suspended" | "revoked";

type Restaurant = {
  id: string;
  name: string;
  legal_name: string | null;
  description: string | null;
  city: string;
  business_email: string | null;
  business_phone: string | null;
  website_url: string | null;
  category: string | null;
  status: RestaurantStatus;
  verification_status: VerificationStatus;
};

type Outlet = {
  id: string;
  restaurant_id: string;
  name: string;
  city: string;
  address_line1: string | null;
  address_line2: string | null;
  landmark: string | null;
  state: string | null;
  postal_code: string | null;
  country: string;
  phone: string | null;
  email: string | null;
  is_primary: boolean;
};

function getApplicationState(
  restaurantStatus: RestaurantStatus,
  verificationStatus: VerificationStatus,
  membershipStatus: MembershipStatus
) {
  if (membershipStatus === "suspended" || membershipStatus === "revoked") {
    return {
      title: "Membership restricted",
      description:
        "Your access to this restaurant workspace is currently restricted. Contact Apronly support for assistance.",
      tone: "warning",
      approved: false,
    };
  }

  if (membershipStatus === "invited") {
    return {
      title: "Invitation pending",
      description: "Your restaurant membership has not been activated yet.",
      tone: "neutral",
      approved: false,
    };
  }

  if (restaurantStatus === "suspended") {
    return {
      title: "Restaurant suspended",
      description:
        "This restaurant workspace is currently suspended. Contact Apronly support for further information.",
      tone: "warning",
      approved: false,
    };
  }

  if (
    restaurantStatus === "rejected" ||
    verificationStatus === "rejected"
  ) {
    return {
      title: "Application not approved",
      description:
        "Your restaurant application was not approved. Please contact Apronly support for information about the decision and possible next steps.",
      tone: "danger",
      approved: false,
    };
  }

  if (
    restaurantStatus === "active" &&
    verificationStatus === "verified" &&
    membershipStatus === "active"
  ) {
    return {
      title: "Restaurant approved",
      description:
        "Your restaurant has been verified and approved. Your business, outlet management and hiring workspace are available below.",
      tone: "success",
      approved: true,
    };
  }

  return {
    title: "Application under review",
    description:
      "Your restaurant application has been received. Apronly will review your details before enabling hiring capabilities.",
    tone: "neutral",
    approved: false,
  };
}

function StatusCard({
  title,
  description,
  tone,
}: {
  title: string;
  description: string;
  tone: string;
}) {
  const toneClasses = {
    success: "border-sage/30 bg-sage/10",
    warning: "border-accent/30 bg-accent/10",
    danger: "border-danger/20 bg-danger/5",
    neutral: "border-border bg-surface-muted",
  };

  const selectedTone =
    toneClasses[tone as keyof typeof toneClasses] ?? toneClasses.neutral;

  return (
    <div className={"rounded-2xl border p-5 " + selectedTone}>
      <h3 className="font-semibold">{title}</h3>
      <p className="mt-2 text-sm leading-6 text-muted">{description}</p>
    </div>
  );
}

async function RestaurantDashboardContent() {
  await connection();

  const access = await requireAnyRole(["restaurant_staff"]);
  const supabase = await createClient();

  const { data: memberships, error: membershipError } = await supabase
    .from("restaurant_memberships")
    .select("id, restaurant_id, role, status")
    .eq("user_id", access.userId);

  if (membershipError) {
    console.error("Restaurant membership lookup failed:", {
      code: membershipError.code,
    });
  }

  const membershipRecords = memberships ?? [];

  const activeRestaurantIds = membershipRecords
    .filter((membership) => membership.status === "active")
    .map((membership) => membership.restaurant_id);

  let restaurantRecords: Restaurant[] = [];
  let outletRecords: Outlet[] = [];
  let restaurantQueryFailed = false;
  let outletQueryFailed = false;

  if (activeRestaurantIds.length > 0) {
    const [restaurantResult, outletResult] = await Promise.all([
      supabase
        .from("restaurants")
        .select(
          "id, name, legal_name, description, city, business_email, business_phone, website_url, category, status, verification_status"
        )
        .in("id", activeRestaurantIds),

      supabase
        .from("restaurant_outlets")
        .select(
          "id, restaurant_id, name, city, address_line1, address_line2, landmark, state, postal_code, country, phone, email, is_primary"
        )
        .in("restaurant_id", activeRestaurantIds)
        .order("is_primary", { ascending: false })
        .order("name", { ascending: true }),
    ]);

    if (restaurantResult.error) {
      console.error("Restaurant details lookup failed:", {
        code: restaurantResult.error.code,
      });
      restaurantQueryFailed = true;
    } else {
      restaurantRecords = (restaurantResult.data ?? []) as Restaurant[];
    }

    if (outletResult.error) {
      console.error("Restaurant outlet lookup failed:", {
        code: outletResult.error.code,
      });
      outletQueryFailed = true;
    } else {
      outletRecords = (outletResult.data ?? []) as Outlet[];
    }
  }

  const restaurantById = new Map(
    restaurantRecords.map((restaurant) => [restaurant.id, restaurant])
  );

  const outletsByRestaurant = new Map<string, Outlet[]>();

  for (const outlet of outletRecords) {
    const existing = outletsByRestaurant.get(outlet.restaurant_id) ?? [];
    existing.push(outlet);
    outletsByRestaurant.set(outlet.restaurant_id, existing);
  }

  const membershipLookupFailed = Boolean(membershipError);

  return (
    <main className="min-h-screen bg-background">
      <div className="mx-auto max-w-6xl px-5 py-8 sm:px-8 lg:py-12">
        <header className="flex flex-wrap items-center justify-between gap-4 border-b border-border pb-6">
          <Link
            href="/"
            className="flex items-center gap-2 text-xl font-semibold tracking-tight"
          >
            <span className="flex h-9 w-9 items-center justify-center rounded-xl bg-primary text-sm text-white">
              A
            </span>
            Apronly
          </Link>

          <div className="flex items-center gap-3">
            <span className="hidden text-sm text-muted sm:inline">
              Restaurant workspace
            </span>

            <form action="/auth/signout" method="post">
              <button
                type="submit"
                className="rounded-xl border border-border px-4 py-2 text-sm font-medium transition hover:bg-surface-muted"
              >
                Sign out
              </button>
            </form>
          </div>
        </header>

        <section className="py-12 sm:py-16">
          <p className="text-sm font-medium text-sage">
            RESTAURANT PARTNERSHIP
          </p>

          <h1 className="mt-3 text-3xl font-semibold tracking-tight sm:text-4xl">
            Your restaurant workspace
          </h1>

          <p className="mt-4 max-w-2xl text-sm leading-7 text-muted">
            Manage your restaurant business information, physical outlets and
            hiring requirements through Apronly.
          </p>
        </section>

        {membershipLookupFailed ? (
          <StatusCard
            title="Unable to load your workspace"
            description="We couldn't retrieve your restaurant membership details. Please refresh the page or contact Apronly support if the problem continues."
            tone="danger"
          />
        ) : membershipRecords.length === 0 ? (
          <StatusCard
            title="Restaurant membership unavailable"
            description="Your account currently has no restaurant membership. Please contact Apronly support to resolve your account access."
            tone="warning"
          />
        ) : (
          <section className="space-y-8">
            <div>
              <h2 className="text-xl font-semibold">Your restaurants</h2>
              <p className="mt-2 text-sm text-muted">
                Application, verification and management for your restaurant
                memberships.
              </p>
            </div>

            {membershipRecords.map((membership) => {
              const restaurant = restaurantById.get(
                membership.restaurant_id
              );

              if (membership.status === "active" && restaurantQueryFailed) {
                return (
                  <article
                    key={membership.id}
                    className="rounded-3xl border border-border bg-surface p-6 sm:p-8"
                  >
                    <StatusCard
                      title="Unable to load restaurant details"
                      description="Your membership was found, but we couldn't retrieve the restaurant's approval details. Management access remains unavailable until this is resolved."
                      tone="warning"
                    />
                  </article>
                );
              }

              if (membership.status === "active" && !restaurant) {
                return (
                  <article
                    key={membership.id}
                    className="rounded-3xl border border-border bg-surface p-6 sm:p-8"
                  >
                    <StatusCard
                      title="Restaurant details unavailable"
                      description="Your active membership was found, but we couldn't retrieve the restaurant record. Contact Apronly support."
                      tone="warning"
                    />
                  </article>
                );
              }

              const applicationState = restaurant
                ? getApplicationState(
                    restaurant.status,
                    restaurant.verification_status,
                    membership.status as MembershipStatus
                  )
                : membership.status === "active"
                  ? null
                  : {
                      title: "Membership restricted",
                      description:
                        "This membership is not active. Restaurant details are unavailable until access is restored.",
                      tone: "warning",
                      approved: false,
                    };

              if (!applicationState) return null;

              const canEdit =
                applicationState.approved && membership.role === "owner";

              const canEditPendingApplication =
                restaurant?.status === "pending" &&
                restaurant.verification_status === "unverified" &&
                membership.status === "active" &&
                membership.role === "owner";

              return (
                <article
                  key={membership.id}
                  className="rounded-3xl border border-border bg-surface p-6 sm:p-8"
                >
                  <div className="flex flex-wrap items-start justify-between gap-4">
                    <div>
                      <h3 className="text-xl font-semibold">
                        {restaurant?.name ?? "Restaurant workspace"}
                      </h3>

                      {restaurant && (
                        <p className="mt-1 text-sm text-muted">
                          {restaurant.city}
                        </p>
                      )}
                    </div>

                    <span className="rounded-full border border-border bg-surface-muted px-3 py-1 text-xs font-medium capitalize">
                      {membership.role}
                    </span>
                  </div>

                  <div className="mt-6">
                    <StatusCard
                      title={applicationState.title}
                      description={applicationState.description}
                      tone={applicationState.tone}
                    />

                    {canEditPendingApplication && (
                      <div className="mt-4">
                        <Link
                          href="/partner/edit"
                          className="inline-flex items-center justify-center rounded-xl bg-primary px-5 py-3 text-sm font-semibold text-white transition hover:opacity-90"
                        >
                          Edit application
                        </Link>
                      </div>
                    )}
                  </div>

                  {restaurant && (
                    <div className="mt-6 grid gap-4 border-t border-border pt-5 sm:grid-cols-2">
                      <div>
                        <p className="text-xs font-medium uppercase tracking-wider text-muted">
                          Restaurant status
                        </p>
                        <p className="mt-2 text-sm font-medium capitalize">
                          {restaurant.status}
                        </p>
                      </div>

                      <div>
                        <p className="text-xs font-medium uppercase tracking-wider text-muted">
                          Verification
                        </p>
                        <p className="mt-2 text-sm font-medium capitalize">
                          {restaurant.verification_status}
                        </p>
                      </div>
                    </div>
                  )}

                  {applicationState.approved && restaurant && (
                    <>
                      {outletQueryFailed ? (
                        <div className="mt-6">
                          <StatusCard
                            title="Unable to load outlets"
                            description="Your restaurant is approved, but outlet information could not be retrieved. Refresh the page or contact support."
                            tone="warning"
                          />
                        </div>
                      ) : (
                        <RestaurantManagement
                          restaurant={restaurant}
                          outlets={
                            outletsByRestaurant.get(restaurant.id) ?? []
                          }
                          canEdit={canEdit}
                        />
                      )}

                      <HiringManagement restaurantId={restaurant.id} />
                    </>
                  )}
                </article>
              );
            })}
          </section>
        )}

        <footer className="mt-12 border-t border-border py-6 text-xs text-muted">
          Apronly · Hospitality recruitment, thoughtfully connected.
        </footer>
      </div>
    </main>
  );
}

"use client";

import { useState } from "react";
import { useRouter } from "next/navigation";
import { createClient } from "@/lib/supabase/client";

type Restaurant = {
  id: string;
  name: string;
  legal_name: string | null;
  description: string | null;
  business_email: string | null;
  business_phone: string | null;
  website_url: string | null;
  category: string | null;
};

type Outlet = {
  id: string;
  name: string;
  city: string;
  address_line1: string | null;
  address_line2: string | null;
  landmark: string | null;
  state: string | null;
  postal_code: string | null;
  country: string;
  phone: string | null;
  email: string | null;
  is_primary: boolean;
};

type Props = {
  restaurant: Restaurant;
  outlets: Outlet[];
  canEdit: boolean;
};

const inputClass =
  "w-full rounded-xl border border-border bg-background px-4 py-3 text-sm outline-none focus:border-accent disabled:opacity-60";

const labelClass = "mb-2 block text-sm font-medium";

const categories = [
  "Fine Dining",
  "Casual Dining",
  "Cafe",
  "Quick Service Restaurant",
  "Bakery",
  "Cloud Kitchen",
  "Bar and Lounge",
  "Hotel Restaurant",
  "Food Court",
  "Other",
];

const emptyOutlet = {
  name: "",
  city: "",
  address_line1: "",
  address_line2: "",
  landmark: "",
  state: "",
  postal_code: "",
  country: "India",
  phone: "",
  email: "",
  is_primary: false,
};

function normalize(value: string) {
  return value.trim().replace(/\s+/g, " ");
}

function Field({
  label,
  value,
  onChange,
  required = false,
  type = "text",
  maxLength,
  placeholder,
}: {
  label: string;
  value: string;
  onChange: (value: string) => void;
  required?: boolean;
  type?: string;
  maxLength?: number;
  placeholder?: string;
}) {
  return (
    <label className="block">
      <span className={labelClass}>
        {label}
        {required && <span className="ml-1 text-danger">*</span>}
      </span>
      <input
        className={inputClass}
        type={type}
        value={value}
        onChange={(event) => onChange(event.target.value)}
        required={required}
        maxLength={maxLength}
        placeholder={placeholder}
      />
    </label>
  );
}

export default function RestaurantManagement({
  restaurant,
  outlets: initialOutlets,
  canEdit,
}: Props) {
  const router = useRouter();

  const [profile, setProfile] = useState({
    name: restaurant.name ?? "",
    legalName: restaurant.legal_name ?? "",
    description: restaurant.description ?? "",
    email: restaurant.business_email ?? "",
    phone: restaurant.business_phone ?? "",
    website: restaurant.website_url ?? "",
    category: restaurant.category ?? "",
  });

  const [outlets, setOutlets] = useState(initialOutlets);
  const [editingProfile, setEditingProfile] = useState(false);
  const [editingOutlet, setEditingOutlet] = useState<string | null>(null);
  const [outletForm, setOutletForm] = useState(emptyOutlet);
  const [showOutletForm, setShowOutletForm] = useState(false);

  const [savingProfile, setSavingProfile] = useState(false);
  const [savingOutlet, setSavingOutlet] = useState(false);
  const [error, setError] = useState("");
  const [success, setSuccess] = useState("");

  function updateProfile(key: keyof typeof profile, value: string) {
    setProfile((current) => ({ ...current, [key]: value }));
  }

  function updateOutlet(
    key: keyof typeof emptyOutlet,
    value: string | boolean,
  ) {
    setOutletForm((current) => ({ ...current, [key]: value }));
  }

  async function saveProfile(event: React.FormEvent<HTMLFormElement>) {
    event.preventDefault();
    setError("");
    setSuccess("");

    if (!canEdit || savingProfile) return;

    const clean = {
      name: normalize(profile.name),
      legalName: normalize(profile.legalName),
      description: profile.description.trim(),
      email: profile.email.trim().toLowerCase(),
      phone: profile.phone.trim(),
      website: profile.website.trim(),
      category: normalize(profile.category),
    };

    if (clean.name.length < 2) {
      setError("Restaurant name must contain at least 2 characters.");
      return;
    }

    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(clean.email)) {
      setError("Enter a valid business email address.");
      return;
    }

    if (!/^[+0-9() .-]{7,30}$/.test(clean.phone)) {
      setError("Enter a valid business phone number.");
      return;
    }

    if (
      clean.website &&
      !/^https?:\/\/.+/i.test(clean.website)
    ) {
      setError("Website must start with http:// or https://.");
      return;
    }

    if (!clean.category) {
      setError("Select a restaurant category.");
      return;
    }

    setSavingProfile(true);

    try {
      const supabase = createClient();

      const { error: saveError } = await supabase.rpc(
        "update_restaurant_business_profile",
        {
          p_restaurant_id: restaurant.id,
          p_name: clean.name,
          p_legal_name: clean.legalName || null,
          p_description: clean.description || null,
          p_business_email: clean.email,
          p_business_phone: clean.phone,
          p_website_url: clean.website || null,
          p_category: clean.category,
        },
      );

      if (saveError) throw saveError;

      setProfile({
        name: clean.name,
        legalName: clean.legalName,
        description: clean.description,
        email: clean.email,
        phone: clean.phone,
        website: clean.website,
        category: clean.category,
      });

      setEditingProfile(false);
      setSuccess("Business profile updated successfully.");
      router.refresh();
    } catch (err) {
      console.error("Restaurant profile update failed:", err);
      setError(
        err instanceof Error
          ? err.message
          : "Unable to update restaurant profile.",
      );
    } finally {
      setSavingProfile(false);
    }
  }

  function startNewOutlet() {
    setError("");
    setSuccess("");
    setEditingOutlet(null);
    setOutletForm({
      ...emptyOutlet,
      is_primary: outlets.length === 0,
    });
    setShowOutletForm(true);
  }

  function startEditOutlet(outlet: Outlet) {
    setError("");
    setSuccess("");
    setEditingOutlet(outlet.id);
    setOutletForm({
      name: outlet.name,
      city: outlet.city,
      address_line1: outlet.address_line1 ?? "",
      address_line2: outlet.address_line2 ?? "",
      landmark: outlet.landmark ?? "",
      state: outlet.state ?? "",
      postal_code: outlet.postal_code ?? "",
      country: outlet.country,
      phone: outlet.phone ?? "",
      email: outlet.email ?? "",
      is_primary: outlet.is_primary,
    });
    setShowOutletForm(true);
  }

  async function saveOutlet(event: React.FormEvent<HTMLFormElement>) {
    event.preventDefault();
    setError("");
    setSuccess("");

    if (!canEdit || savingOutlet) return;

    const clean = {
      name: normalize(outletForm.name),
      city: normalize(outletForm.city),
      address_line1: normalize(outletForm.address_line1),
      address_line2: normalize(outletForm.address_line2),
      landmark: normalize(outletForm.landmark),
      state: normalize(outletForm.state),
      postal_code: normalize(outletForm.postal_code),
      country: normalize(outletForm.country),
      phone: outletForm.phone.trim(),
      email: outletForm.email.trim().toLowerCase(),
      is_primary: outletForm.is_primary,
    };

    if (clean.name.length < 2 || clean.city.length < 2) {
      setError("Enter a valid outlet name and city.");
      return;
    }

    if (clean.address_line1.length < 5) {
      setError("Enter the complete outlet address.");
      return;
    }

    if (!clean.state || !clean.postal_code || !clean.country) {
      setError("State, postal code and country are required.");
      return;
    }

    if (
      clean.phone &&
      !/^[+0-9() .-]{7,30}$/.test(clean.phone)
    ) {
      setError("Enter a valid outlet phone number.");
      return;
    }

    if (
      clean.email &&
      !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(clean.email)
    ) {
      setError("Enter a valid outlet email address.");
      return;
    }

    setSavingOutlet(true);

    try {
      const supabase = createClient();

      const { data: savedId, error: saveError } = await supabase.rpc(
        "save_restaurant_outlet",
        {
          p_restaurant_id: restaurant.id,
          p_outlet_id: editingOutlet,
          p_name: clean.name,
          p_city: clean.city,
          p_address_line1: clean.address_line1,
          p_address_line2: clean.address_line2 || null,
          p_landmark: clean.landmark || null,
          p_state: clean.state,
          p_postal_code: clean.postal_code,
          p_country: clean.country,
          p_phone: clean.phone || null,
          p_email: clean.email || null,
          p_is_primary: clean.is_primary,
        },
      );

      if (saveError) throw saveError;

      if (!savedId) {
        throw new Error("The outlet was not saved. Please try again.");
      }

      const updatedOutlet: Outlet = {
        id: savedId,
        ...clean,
      };

      setOutlets((current) => {
        const next = editingOutlet
          ? current.map((outlet) =>
              outlet.id === savedId ? updatedOutlet : outlet,
            )
          : [...current, updatedOutlet];

        if (clean.is_primary) {
          return next.map((outlet) => ({
            ...outlet,
            is_primary: outlet.id === savedId,
          }));
        }

        return next;
      });

      setShowOutletForm(false);
      setEditingOutlet(null);
      setSuccess(
        editingOutlet
          ? "Outlet updated successfully."
          : "New outlet added successfully.",
      );

      router.refresh();
    } catch (err) {
      console.error("Restaurant outlet save failed:", err);
      setError(
        err instanceof Error
          ? err.message
          : "Unable to save outlet.",
      );
    } finally {
      setSavingOutlet(false);
    }
  }

  return (
    <section className="mt-8 space-y-6">
      {(error || success) && (
        <div
          role="status"
          className={`rounded-xl border p-4 text-sm ${
            error
              ? "border-danger/30 bg-danger/5 text-danger"
              : "border-sage/30 bg-sage/10 text-foreground"
          }`}
        >
          {error || success}
        </div>
      )}

      {/* BUSINESS PROFILE */}

      <article className="rounded-3xl border border-border bg-surface p-6 sm:p-8">
        <div className="flex flex-wrap items-start justify-between gap-4">
          <div>
            <p className="text-xs font-semibold uppercase tracking-wider text-sage">
              Business information
            </p>
            <h3 className="mt-2 text-xl font-semibold">
              Restaurant profile
            </h3>
            <p className="mt-2 text-sm text-muted">
              Manage the business details associated with your Apronly account.
            </p>
          </div>

          {canEdit && !editingProfile && (
            <button
              type="button"
              onClick={() => {
                setError("");
                setSuccess("");
                setEditingProfile(true);
              }}
              className="rounded-xl border border-border px-4 py-2 text-sm font-medium hover:bg-surface-muted"
            >
              Edit profile
            </button>
          )}
        </div>

        {editingProfile ? (
          <form onSubmit={saveProfile} className="mt-7 space-y-5">
            <div className="grid gap-5 sm:grid-cols-2">
              <Field
                label="Restaurant name"
                value={profile.name}

import { Suspense } from "react";
import { connection } from "next/server";
import { redirect } from "next/navigation";
import { requireAnyRole } from "@/lib/auth/authorization";
import { createClient } from "@/lib/supabase/server";
import RestaurantApplicationForm from "@/app/onboarding/restaurant/restaurant-application-form";

async function EditRestaurantApplicationContent() {
  await connection();

  const access = await requireAnyRole(["restaurant_staff"]);
  const supabase = await createClient();

  const { data: membership, error: membershipError } = await supabase
    .from("restaurant_memberships")
    .select("restaurant_id, role, status")
    .eq("user_id", access.userId)
    .eq("role", "owner")
    .eq("status", "active")
    .limit(1)
    .maybeSingle();

  if (membershipError || !membership) {
    redirect("/partner");
  }

  const { data: restaurant, error: restaurantError } = await supabase
    .from("restaurants")
    .select(
      "id, name, legal_name, description, business_email, business_phone, website_url, category, status, verification_status"
    )
    .eq("id", membership.restaurant_id)
    .maybeSingle();

  if (
    restaurantError ||
    !restaurant ||
    restaurant.status !== "pending" ||
    restaurant.verification_status !== "unverified"
  ) {
    redirect("/partner");
  }

  const { data: outlet, error: outletError } = await supabase
    .from("restaurant_outlets")
    .select(
      "name, address_line1, address_line2, landmark, city, state, postal_code, country, phone, email"
    )
    .eq("restaurant_id", restaurant.id)
    .eq("is_primary", true)
    .maybeSingle();

  if (outletError || !outlet) {
    redirect("/partner");
  }

  return (
    <RestaurantApplicationForm
      isEditing
      initialValues={{
        name: restaurant.name ?? "",
        legalName: restaurant.legal_name ?? "",
        description: restaurant.description ?? "",
        businessEmail: restaurant.business_email ?? "",
        businessPhone: restaurant.business_phone ?? "",
        websiteUrl: restaurant.website_url ?? "",
        category: restaurant.category ?? "",
        outletName: outlet.name ?? "",
        addressLine1: outlet.address_line1 ?? "",
        addressLine2: outlet.address_line2 ?? "",
        landmark: outlet.landmark ?? "",
        city: outlet.city ?? "",
        state: outlet.state ?? "",
        postalCode: outlet.postal_code ?? "",
        country: outlet.country ?? "India",
        outletPhone: outlet.phone ?? "",
        outletEmail: outlet.email ?? "",
      }}
    />
  );
}

export default function EditRestaurantApplicationPage() {
  return (
    <Suspense
      fallback={
        <main className="flex min-h-screen items-center justify-center bg-background">
          <p className="text-sm text-muted">Loading your application...</p>
        </main>
      }
    >
      <EditRestaurantApplicationContent />
    </Suspense>
  );
}
