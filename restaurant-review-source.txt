"use client";

import { useMemo, useState } from "react";
import type { ReactNode } from "react";
import { useRouter } from "next/navigation";
import { createClient } from "@/lib/supabase/client";

interface RestaurantOutlet {
  id: string;
  name: string;
  city: string;
  address_line1: string | null;
  address_line2: string | null;
  landmark: string | null;
  state: string | null;
  postal_code: string | null;
  country: string;
  phone: string | null;
  email: string | null;
  is_primary: boolean;
}

export interface RestaurantApplication {
  id: string;
  name: string;
  legal_name: string | null;
  slug: string;
  description: string | null;
  city: string;
  business_email: string | null;
  business_phone: string | null;
  website_url: string | null;
  category: string | null;
  status: string;
  verification_status: string;
  review_status: string;
  review_note: string | null;
  reviewed_by: string | null;
  reviewed_at: string | null;
  created_at: string;
  outlets: RestaurantOutlet[];
}

interface RestaurantReviewProps {
  applications: RestaurantApplication[];
  loadError?: string | null;
}

type ReviewDecision = "approved" | "rejected" | "changes_requested";

function formatDate(value: string) {
  return new Intl.DateTimeFormat("en-IN", {
    dateStyle: "medium",
    timeStyle: "short",
  }).format(new Date(value));
}

function Detail({
  label,
  children,
}: {
  label: string;
  children: ReactNode;
}) {
  return (
    <div className="min-w-0">
      <p className="text-[11px] font-semibold uppercase tracking-wider text-muted">
        {label}
      </p>
      <div className="mt-1 break-words text-sm leading-6 text-foreground">
        {children || "Not provided"}
      </div>
    </div>
  );
}

function StatusPill({ children }: { children: ReactNode }) {
  return (
    <span className="inline-flex rounded-full border border-amber-500/30 bg-amber-500/10 px-2.5 py-1 text-[11px] font-semibold text-amber-700">
      {children}
    </span>
  );
}

function RestaurantApplicationDetails({
  application,
  onBack,
  showBackButton,
}: {
  application: RestaurantApplication;
  onBack: () => void;
  showBackButton: boolean;
}) {
  const router = useRouter();
  const [note, setNote] = useState("");
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState("");
  const [expanded, setExpanded] = useState(false);

  const primaryOutlet =
    application.outlets?.find((outlet) => outlet.is_primary) ??
    application.outlets?.[0];

  const outletAddress = primaryOutlet
    ? [
        primaryOutlet.address_line1,
        primaryOutlet.address_line2,
        primaryOutlet.landmark,
        primaryOutlet.city,
        primaryOutlet.state,
        primaryOutlet.postal_code,
        primaryOutlet.country,
      ]
        .filter(Boolean)
        .join(", ")
    : "No outlet information available";

  async function review(decision: ReviewDecision) {
    const trimmedNote = note.trim();

    if (
      (decision === "rejected" || decision === "changes_requested") &&
      trimmedNote.length < 10
    ) {
      setError("Please provide a reason of at least 10 characters.");
      return;
    }

    const confirmed = window.confirm(
      decision === "approved"
        ? `Approve and verify ${application.name}? This enables its recruitment workspace.`
        : decision === "rejected"
          ? `Reject the application from ${application.name}?`
          : `Request changes from ${application.name}?`,
    );

    if (!confirmed) return;

    setBusy(true);
    setError("");

    try {
      const supabase = createClient();

      const { error: reviewError } = await supabase.rpc(
        "review_restaurant_application",
        {
          p_restaurant_id: application.id,
          p_decision: decision,
          p_review_note: trimmedNote || null,
        },
      );

      if (reviewError) throw reviewError;

      router.refresh();
    } catch (err) {
      console.error("Restaurant application review failed:", err);
      setError(
        err instanceof Error
          ? err.message
          : "Unable to submit the review. Please try again.",
      );
    } finally {
      setBusy(false);
    }
  }

  return (
    <div className="flex h-full min-h-0 flex-col">
      <div className="min-h-0 flex-1 overflow-y-auto overscroll-contain">
        <div className="border-b border-border px-5 py-5 sm:px-7">
          {showBackButton && (
            <button
              type="button"
              onClick={onBack}
              className="mb-4 inline-flex items-center gap-2 text-sm font-medium text-muted hover:text-foreground lg:hidden"
            >
              <span aria-hidden="true">←</span>
              Back to applications
            </button>
          )}

          <div className="flex flex-wrap items-start justify-between gap-3">
            <div className="min-w-0">
              <p className="text-xs font-semibold uppercase tracking-wider text-sage">
                {application.category || "Restaurant"} · {application.city}
              </p>
              <h3 className="mt-2 text-2xl font-semibold tracking-tight">
                {application.name}
              </h3>
              <p className="mt-1 text-xs text-muted">
                Submitted {formatDate(application.created_at)}
              </p>
            </div>
            <StatusPill>Pending verification</StatusPill>
          </div>
        </div>

        <div className="space-y-7 px-5 py-6 sm:px-7">
          <section>
            <h4 className="mb-4 text-sm font-semibold">Business information</h4>
            <div className="grid gap-x-6 gap-y-5 sm:grid-cols-2">
              <Detail label="Legal business name">
                {application.legal_name}
              </Detail>
              <Detail label="Business category">
                {application.category}
              </Detail>
              <Detail label="Business email">
                {application.business_email}
              </Detail>
              <Detail label="Business phone">
                {application.business_phone}
              </Detail>
              <Detail label="Website">
                {application.website_url ? (
                  <a
                    href={application.website_url}
                    target="_blank"
                    rel="noreferrer"
                    className="text-sage underline underline-offset-4"
                  >
                    {application.website_url}
                  </a>
                ) : (
                  "Not provided"
                )}
              </Detail>
              <Detail label="Registered city">{application.city}</Detail>
            </div>

            {application.description && (
              <div className="mt-5 border-t border-border pt-5">
                <Detail label="Business description">
                  <p className="whitespace-pre-wrap">
                    {application.description}
                  </p>
                </Detail>
              </div>
            )}
          </section>

          <section className="border-t border-border pt-6">
            <div className="flex flex-wrap items-center justify-between gap-2">
              <h4 className="text-sm font-semibold">Outlet information</h4>
              <span className="text-xs text-muted">
                {application.outlets?.length ?? 0} outlet(s) registered
              </span>
            </div>

            {primaryOutlet ? (
              <div className="mt-4 rounded-2xl border border-border p-4">
                <div className="mb-4 flex items-center justify-between gap-2">
                  <p className="text-sm font-semibold">{primaryOutlet.name}</p>
                  <span className="text-[10px] font-semibold uppercase tracking-wide text-sage">
                    Primary outlet
                  </span>
                </div>
                <div className="grid gap-4 sm:grid-cols-2">
                  <Detail label="Outlet phone">
                    {primaryOutlet.phone}
                  </Detail>
                  <Detail label="Outlet email">
                    {primaryOutlet.email}
                  </Detail>
                  <div className="sm:col-span-2">
                    <Detail label="Complete address">
                      {outletAddress}
                    </Detail>
                  </div>
                </div>
              </div>
            ) : (
              <p className="mt-3 text-sm text-muted">
                No outlet information was found.
              </p>
            )}

            {application.outlets && application.outlets.length > 1 && (
              <div className="mt-4 space-y-3">
                <p className="text-xs font-semibold uppercase tracking-wide text-muted">
                  Additional outlets
                </p>
                {application.outlets
                  .filter((outlet) => !outlet.is_primary)
                  .map((outlet) => (
                    <div
                      key={outlet.id}
                      className="rounded-2xl border border-border p-4"
                    >
                      <p className="text-sm font-medium">{outlet.name}</p>
                      <p className="mt-1 text-sm text-muted">
                        {[
                          outlet.address_line1,
                          outlet.address_line2,
                          outlet.city,
                          outlet.state,
                          outlet.postal_code,
                          outlet.country,
                        ]
                          .filter(Boolean)
                          .join(", ")}
                      </p>
                      <p className="mt-2 text-xs text-muted">
                        {outlet.phone || "No outlet phone provided"}
                        {outlet.email ? ` · ${outlet.email}` : ""}
                      </p>
                    </div>
                  ))}
              </div>
            )}
          </section>

          <section className="border-t border-border pt-6">
            <button
              type="button"
              onClick={() => setExpanded((value) => !value)}
              aria-expanded={expanded}
              className="flex w-full items-center justify-between text-left text-sm font-semibold hover:text-sage"
            >
              Technical and system details
              <span className="text-muted">{expanded ? "−" : "+"}</span>
            </button>

            {expanded && (
              <div className="mt-4 grid gap-4 rounded-2xl bg-surface-muted p-4 sm:grid-cols-2">
                <Detail label="Application ID">
                  <span className="font-mono text-xs">{application.id}</span>
                </Detail>
                <Detail label="Restaurant slug">
                  {application.slug}
                </Detail>
                <Detail label="Restaurant status">
                  {application.status}
                </Detail>
                <Detail label="Verification status">
                  {application.verification_status}
                </Detail>
                <Detail label="Review status">
                  {application.review_status}
                </Detail>
                <Detail label="Last review note">
                  {application.review_note}
                </Detail>
              </div>
            )}
          </section>
        </div>
      </div>

      <div className="shrink-0 border-t border-border bg-surface px-5 py-4 sm:px-7">
        <label
          htmlFor={`restaurant-note-${application.id}`}
          className="block text-sm font-semibold"
        >
          Administrator review note
        </label>
        <p className="mt-1 text-xs text-muted">
          Required for rejection or change requests. Minimum 10 characters.
        </p>

        <textarea
          id={`restaurant-note-${application.id}`}
          value={note}
          onChange={(event) => setNote(event.target.value)}
          maxLength={2000}
          rows={2}
          disabled={busy}
          placeholder="Enter verification feedback or explain your decision..."
          className="mt-3 w-full resize-y rounded-xl border border-border bg-background px-3 py-2.5 text-sm outline-none transition focus:border-sage disabled:opacity-50"
        />

        <div className="mt-1 text-right text-[11px] text-muted">
          {note.length}/2000
        </div>

        {error && (
          <p role="alert" className="mt-2 text-sm text-red-600">
            {error}
          </p>
        )}

        <div className="mt-3 grid grid-cols-1 gap-2 sm:grid-cols-3">
          <button
            type="button"
            disabled={busy}
            onClick={() => void review("approved")}
            className="rounded-xl bg-primary px-3 py-2.5 text-sm font-semibold text-white transition hover:opacity-90 disabled:cursor-not-allowed disabled:opacity-50"
          >
            {busy ? "Processing..." : "Approve & verify"}
          </button>

          <button
            type="button"
            disabled={busy}
            onClick={() => void review("changes_requested")}
            className="rounded-xl border border-border px-3 py-2.5 text-sm font-semibold transition hover:bg-surface-muted disabled:cursor-not-allowed disabled:opacity-50"
          >
            Request changes
          </button>

          <button
            type="button"
            disabled={busy}
            onClick={() => void review("rejected")}
            className="rounded-xl border border-red-500/30 px-3 py-2.5 text-sm font-semibold text-red-600 transition hover:bg-red-500/5 disabled:cursor-not-allowed disabled:opacity-50"
          >
            Reject application
          </button>
        </div>
      </div>
    </div>
  );
}

export default function RestaurantReview({
  applications,
  loadError,
}: RestaurantReviewProps) {
  const [selectedId, setSelectedId] = useState<string | null>(
    applications[0]?.id ?? null,
  );
  const [search, setSearch] = useState("");
  const [cityFilter, setCityFilter] = useState("all");
  const [sortOrder, setSortOrder] = useState<"oldest" | "newest">("oldest");
  const [mobileDetailOpen, setMobileDetailOpen] = useState(false);

  const cities = useMemo(
    () =>
      Array.from(new Set(applications.map((item) => item.city).filter(Boolean)))
        .sort((a, b) => a.localeCompare(b)),
    [applications],
  );

  const filteredApplications = useMemo(() => {
    const query = search.trim().toLowerCase();

    return applications
      .filter((application) => {
        const matchesSearch =
          !query ||
          application.name.toLowerCase().includes(query) ||
          application.city.toLowerCase().includes(query) ||
          application.id.toLowerCase().includes(query) ||
          (application.business_email ?? "").toLowerCase().includes(query);

        const matchesCity =
          cityFilter === "all" || application.city === cityFilter;

        return matchesSearch && matchesCity;
      })
      .sort((a, b) => {
        const difference =
          new Date(a.created_at).getTime() -
          new Date(b.created_at).getTime();

        return sortOrder === "oldest" ? difference : -difference;
      });
  }, [applications, search, cityFilter, sortOrder]);

  const selectedApplication =
    filteredApplications.find((item) => item.id === selectedId) ??
    filteredApplications[0] ??
    null;

  function selectApplication(application: RestaurantApplication) {
    setSelectedId(application.id);
    setMobileDetailOpen(true);
  }

  return (
    <section className="mt-12">
      <div className="mb-5 flex flex-wrap items-end justify-between gap-4">
        <div>
          <p className="text-xs font-semibold tracking-[0.16em] text-sage">
            BUSINESS VERIFICATION
          </p>
          <h2 className="mt-2 text-2xl font-semibold tracking-tight">
            Restaurant applications
          </h2>
          <p className="mt-1 text-sm text-muted">
            Review business details and verify restaurant partners.
          </p>
        </div>

        <div className="flex items-center gap-3">
          <div className="rounded-xl border border-border bg-surface px-4 py-2.5">
            <p className="text-[11px] text-muted">Awaiting verification</p>
            <p className="mt-0.5 text-xl font-semibold tabular-nums">
              {applications.length}
            </p>
          </div>
          <div className="rounded-xl border border-border bg-surface px-4 py-2.5">
            <p className="text-[11px] text-muted">Filtered results</p>
            <p className="mt-0.5 text-xl font-semibold tabular-nums">
              {filteredApplications.length}
            </p>
          </div>
        </div>
      </div>

      {loadError ? (
        <div
          role="alert"
          className="rounded-2xl border border-red-500/30 bg-red-500/5 p-5 text-sm text-red-600"
        >
          Unable to load restaurant applications: {loadError}
        </div>
      ) : applications.length === 0 ? (
        <div className="rounded-2xl border border-dashed border-border bg-surface p-10 text-center">
          <h3 className="text-lg font-semibold">
            No pending restaurant applications
          </h3>
          <p className="mt-2 text-sm text-muted">
            New restaurant applications will appear here for verification.
          </p>
        </div>
      ) : (
        <div className="overflow-hidden rounded-2xl border border-border bg-surface">
          <div className="grid min-h-[680px] lg:h-[min(780px,calc(100vh-190px))] lg:grid-cols-[minmax(280px,0.78fr)_minmax(0,1.55fr)]">
            <aside
              className={`min-h-0 border-b border-border lg:border-b-0 lg:border-r ${
                mobileDetailOpen ? "hidden lg:flex" : "flex"
              } flex-col`}
            >
              <div className="shrink-0 space-y-3 border-b border-border p-4">
                <div>
                  <h3 className="text-sm font-semibold">Application queue</h3>
                  <p className="mt-1 text-xs text-muted">
                    Select an application to inspect its details.
                  </p>
                </div>

                <div className="relative">
                  <span className="pointer-events-none absolute left-3 top-1/2 -translate-y-1/2 text-muted">
                    ⌕
                  </span>
                  <input
                    type="search"
                    value={search}
                    onChange={(event) => setSearch(event.target.value)}
                    placeholder="Search name, city or ID..."
                    aria-label="Search restaurant applications"
                    className="w-full rounded-xl border border-border bg-background py-2.5 pl-9 pr-3 text-sm outline-none focus:border-sage"
                  />
                </div>

                <div className="grid grid-cols-2 gap-2">
                  <select
                    value={cityFilter}
                    onChange={(event) => setCityFilter(event.target.value)}
                    aria-label="Filter by city"
                    className="min-w-0 rounded-xl border border-border bg-background px-2.5 py-2 text-xs outline-none focus:border-sage"
                  >
                    <option value="all">All cities</option>
                    {cities.map((city) => (
                      <option key={city} value={city}>
                        {city}
                      </option>
                    ))}
                  </select>

                  <select
                    value={sortOrder}
                    onChange={(event) =>
                      setSortOrder(event.target.value as "oldest" | "newest")
                    }
                    aria-label="Sort applications"
                    className="min-w-0 rounded-xl border border-border bg-background px-2.5 py-2 text-xs outline-none focus:border-sage"
                  >
                    <option value="oldest">Oldest first</option>
                    <option value="newest">Newest first</option>
                  </select>
                </div>
              </div>

              <div className="min-h-0 flex-1 overflow-y-auto overscroll-contain">
                {filteredApplications.length === 0 ? (
                  <div className="p-6 text-center">
                    <p className="text-sm font-medium">No matching applications</p>
                    <p className="mt-1 text-xs text-muted">
                      Try another search term or city.
                    </p>
                  </div>
                ) : (
                  <div className="space-y-1 p-2">
                    {filteredApplications.map((application) => {
                      const selected =
                        selectedApplication?.id === application.id;

                      return (
                        <button
                          key={application.id}
                          type="button"
                          onClick={() => selectApplication(application)}
                          aria-pressed={selected}
                          className={`w-full rounded-xl border p-3 text-left transition ${
                            selected
                              ? "border-sage/50 bg-surface-muted"
                              : "border-transparent hover:border-border hover:bg-surface-muted/70"
                          }`}
                        >
                          <div className="flex items-start justify-between gap-2">
                            <p className="min-w-0 truncate text-sm font-semibold">
                              {application.name}
                            </p>
                            <span
                              className={`mt-1 h-2 w-2 shrink-0 rounded-full ${
                                selected ? "bg-sage" : "bg-amber-500"
                              }`}
                            />
                          </div>

                          <p className="mt-1 text-xs text-muted">
                            {application.category || "Restaurant"} ·{" "}
                            {application.city}
                          </p>

                          <div className="mt-3 flex items-center justify-between gap-2">
                            <span className="text-[11px] text-muted">
                              {formatDate(application.created_at)}
                            </span>
                            <span className="text-[10px] font-medium text-amber-700">
                              Pending
                            </span>
                          </div>
                        </button>
                      );
                    })}
                  </div>
                )}
              </div>

              <div className="shrink-0 border-t border-border px-4 py-3">
                <p className="text-[11px] text-muted">
                  Showing {filteredApplications.length} of {applications.length}{" "}
                  applications
                </p>
              </div>
            </aside>

            <div
              className={`min-h-0 min-w-0 ${
                mobileDetailOpen ? "flex" : "hidden lg:flex"
              } flex-col`}
            >
              {selectedApplication ? (
                <RestaurantApplicationDetails
                  key={selectedApplication.id}
                  application={selectedApplication}
                  onBack={() => setMobileDetailOpen(false)}
                  showBackButton
                />
              ) : (
                <div className="flex flex-1 items-center justify-center p-8 text-center">
                  <div>
                    <h3 className="font-semibold">
                      Select an application
                    </h3>
                    <p className="mt-2 text-sm text-muted">
                      Choose a restaurant from the queue to review its details.
                    </p>
                  </div>
                </div>
              )}
            </div>
          </div>
        </div>
      )}
    </section>
  );
}
